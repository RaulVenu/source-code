
CREATE TABLE customers (
    customer_id INT PRIMARY KEY,
    name VARCHAR(100) NOT NULL
);

CREATE TABLE orders (
    order_id INT PRIMARY KEY,
    customer_id INT NOT NULL,
    ordered_at DATETIME2 NOT NULL,
    status VARCHAR(30) NOT NULL,
    FOREIGN KEY (customer_id) REFERENCES customers(customer_id)
);

CREATE TABLE order_items (
    order_id INT NOT NULL,
    product_id INT NOT NULL,
    quantity INT NOT NULL,
    unit_price DECIMAL(10, 2) NOT NULL,
    FOREIGN KEY (order_id) REFERENCES orders(order_id)
);

CREATE TABLE payments (
    payment_id INT PRIMARY KEY,
    order_id INT NOT NULL,
    amount DECIMAL(10, 2) NOT NULL,
    status VARCHAR(30) NOT NULL,
    paid_at DATETIME2 NULL,
    FOREIGN KEY (order_id) REFERENCES orders(order_id)
);

WITH eligible_orders AS (
    SELECT
        order_id,
        customer_id,
        DATEFROMPARTS(YEAR(ordered_at), MONTH(ordered_at), 1) AS month_start
    FROM orders
    WHERE status <> 'cancelled'
      AND ordered_at >= '2026-01-01'
      AND ordered_at <  '2027-01-01'
),
item_totals AS (
    SELECT
        order_id,
        SUM(quantity * unit_price) AS gross_order_value
    FROM order_items
    GROUP BY order_id
),
payment_totals AS (
    SELECT
        order_id,
        SUM(amount) AS captured_payment_value
    FROM payments
    WHERE status = 'captured'
    GROUP BY order_id
)
SELECT
    c.customer_id,
    c.name,
    eo.month_start,
    COUNT(*) AS order_count,
    SUM(COALESCE(it.gross_order_value, 0)) AS gross_order_value,
    SUM(COALESCE(pt.captured_payment_value, 0)) AS captured_payment_value,
    SUM(COALESCE(it.gross_order_value, 0))
        - SUM(COALESCE(pt.captured_payment_value, 0))
        AS outstanding_value
FROM eligible_orders eo
JOIN customers c
    ON c.customer_id = eo.customer_id
LEFT JOIN item_totals it
    ON it.order_id = eo.order_id
LEFT JOIN payment_totals pt
    ON pt.order_id = eo.order_id
GROUP BY
    c.customer_id,
    c.name,
    eo.month_start
ORDER BY
    c.customer_id,
    eo.month_start;
